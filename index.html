const canvas = document.getElementById("game"); const ctx = canvas.getContext("2d"); canvas.width = 500; canvas.height = 300; // Mundo const gravity = 0.5; // Mario const player = { x: 50, y: 0, w: 20, h: 30, vx: 0, vy: 0, speed: 2.2, jumping: false }; // Plataforma principal (suelo) const ground = { x: 0, y: 260, w: canvas.width, h: 40 }; // Plataforma extra estilo Mario const platform = { x: 200, y: 180, w: 100, h: 20 }; // Inputs const keys = {}; document.addEventListener("keydown", (e) => keys[e.key] = true); document.addEventListener("keyup", (e) => keys[e.key] = false); function movePlayer() { // Movimiento horizontal player.vx = 0; if (keys["ArrowLeft"]) player.vx = -player.speed; if (keys["ArrowRight"]) player.vx = player.speed; // Salto if (keys[" "]) { // Space if (!player.jumping) { player.vy = -10; player.jumping = true; } } // Aplicar movimiento player.x += player.vx; player.vy += gravity; player.y += player.vy; // Limite de pantalla (evita que Mario se vaya) if (player.x < 0) player.x = 0; if (player.x + player.w > canvas.width) player.x = canvas.width - player.w; collide(player, ground); collide(player, platform); } function collide(p, obj) { // Colisión vertical (caída) if ( p.x < obj.x + obj.w && p.x + p.w > obj.x && p.y + p.h > obj.y && p.y + p.h < obj.y + obj.h ) { p.y = obj.y - p.h; p.vy = 0; p.jumping = false; } } function draw() { ctx.clearRect(0, 0, canvas.width, canvas.height); // Fondo ctx.fillStyle = "#9cdcff"; ctx.fillRect(0, 0, canvas.width, canvas.height); // Suelo ctx.fillStyle = "#654321"; ctx.fillRect(ground.x, ground.y, ground.w, ground.h); // Plataforma estilo Mario ctx.fillStyle = "#b5651d"; ctx.fillRect(platform.x, platform.y, platform.w, platform.h); // Mario ctx.fillStyle = "red"; ctx.fillRect(player.x, player.y, player.w, player.h); } function update() { movePlayer(); draw(); requestAnimationFrame(update); } update();
